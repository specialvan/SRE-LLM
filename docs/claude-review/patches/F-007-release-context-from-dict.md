## 目标代码

### `gan_matchmaking/sre/domain.py`（新增 classmethod）

```python
@dataclass
class ReleaseContext:
    service: Service
    candidates: list[ReleaseCandidate]
    # ...existing fields...

    @classmethod
    def from_dict(cls, payload: Mapping[str, Any]) -> "ReleaseContext":
        """Reconstruct a ReleaseContext from a JSON-friendly mapping.

        Expected keys
        -------------
        service : Mapping
            Required. Will be passed to ``Service(**payload['service'])``.
        candidates : Sequence[Mapping]
            Required. Each entry passed to ``ReleaseCandidate(**entry)``.
        dependencies : Sequence[str], optional
            Service ids this release depends on. Default [].
        error_budget_remaining : float, optional
            Value in [0, 1]. Default 1.0.
        freeze_window : bool, optional
            True to force HOLD. Default False.
        telemetry : Mapping[str, float], optional
            Metrics snapshot. Default {}.
        correlation_id : str, optional
            If absent, a uuid4 hex[:16] is generated by ``ReleaseContext``.

        Raises
        ------
        DataError
            If required fields are missing or typed wrong.
        """
        try:
            svc_raw = payload["service"]
            cand_raw = payload["candidates"]
        except KeyError as exc:
            raise DataError(
                f"missing required field in ReleaseContext payload: {exc.args[0]}",
                details={"payload_keys": sorted(payload)},
            ) from exc

        service = Service(**svc_raw)
        candidates = [ReleaseCandidate(**c) for c in cand_raw]
        deps = list(payload.get("dependencies") or [])
        telemetry = dict(payload.get("telemetry") or {})
        return cls(
            service=service,
            candidates=candidates,
            dependencies=deps,
            error_budget_remaining=float(payload.get("error_budget_remaining", 1.0)),
            freeze_window=bool(payload.get("freeze_window", False)),
            telemetry=telemetry,
            correlation_id=payload.get("correlation_id"),
        )
```

```python
def _ctx_from_dict(payload: Mapping[str, Any]) -> ReleaseContext:
```

```python
# tests/test_release_context_serde.py (新建)

from gan_matchmaking import cli
from gan_matchmaking.sre import ReleaseContext


def _sample_payload():
    return {
        "service": {"id": "svc", "mu": 0.95, "sigma": 0.03,
                    "tier": "standard"},
        "candidates": [
            {"id": "c1", "service_id": "svc", "strategy": "canary",
             "canary_fraction": 0.1, "rollback_budget_seconds": 180,
             "expected_success": 0.99},
        ],
        "dependencies": ["dep-a", "dep-b"],
        "error_budget_remaining": 0.7,
        "freeze_window": False,
        "telemetry": {"latency_p99_ms": 120.0},
        "correlation_id": "serde-1",
    }


def test_release_context_from_dict_roundtrip():
    ctx = ReleaseContext.from_dict(_sample_payload())
    assert ctx.service.id == "svc"
    assert ctx.candidates[0].id == "c1"
    assert ctx.dependencies == ["dep-a", "dep-b"]
    assert ctx.error_budget_remaining == 0.7
    assert ctx.freeze_window is False
    assert ctx.telemetry == {"latency_p99_ms": 120.0}
    assert ctx.correlation_id == "serde-1"


def test_cli_helper_delegates_to_release_context():
    payload = _sample_payload()
    ctx_public = ReleaseContext.from_dict(payload)
    ctx_private = cli._ctx_from_dict(payload)  # noqa: SLF001
    assert ctx_public.service.as_dict() == ctx_private.service.as_dict()
    assert [c.id for c in ctx_public.candidates] == \
           [c.id for c in ctx_private.candidates]
    assert ctx_public.dependencies == ctx_private.dependencies
    assert ctx_public.error_budget_remaining == ctx_private.error_budget_remaining


def test_private_helper_not_imported_outside_cli():
    import pathlib
    roots = [
        pathlib.Path("gan_matchmaking/service/app.py"),
        pathlib.Path("tests/test_replay_corpus.py"),
        pathlib.Path("tests/test_replay_export.py"),
    ]
    for p in roots:
        src = p.read_text(encoding="utf-8")
        assert "_ctx_from_dict" not in src, \
            f"{p} still imports/uses private helper"


def test_missing_required_field_raises_data_error():
    import pytest
    from gan_matchmaking.core.errors import DataError
    with pytest.raises(DataError):
        ReleaseContext.from_dict({"service": {"id": "s", "mu": 1, "sigma": 1,
                                              "tier": "standard"}})
        # candidates 缺失


def test_optional_fields_default_values():
    minimal = {
        "service": {"id": "s", "mu": 1.0, "sigma": 0.1, "tier": "standard"},
        "candidates": [],
    }
    ctx = ReleaseContext.from_dict(minimal)
    assert ctx.dependencies == []
    assert ctx.error_budget_remaining == 1.0
    assert ctx.freeze_window is False
    assert ctx.telemetry == {}
```
